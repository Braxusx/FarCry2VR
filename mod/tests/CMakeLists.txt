cmake_minimum_required(VERSION 3.16)
project(fc2vr_tracking_tests LANGUAGES CXX)

enable_testing()
add_executable(hand_roles_tests test_hand_roles.cpp)
target_compile_features(hand_roles_tests PRIVATE cxx_std_17)
target_include_directories(hand_roles_tests PRIVATE ../src)
if(MSVC)
    target_compile_options(hand_roles_tests PRIVATE /W4 /WX)
else()
    target_compile_options(hand_roles_tests PRIVATE -Wall -Wextra -Werror)
endif()
add_test(NAME hand_roles COMMAND hand_roles_tests)
if(WIN32)
    add_executable(weapon_controls_tests test_weapon_controls.cpp)
    target_compile_features(weapon_controls_tests PRIVATE cxx_std_17)
    target_include_directories(weapon_controls_tests PRIVATE ../src ../third_party/openxr_include)
    add_test(NAME weapon_controls COMMAND weapon_controls_tests)
    add_executable(native_input_probe native_input_probe.cpp)
    target_compile_features(native_input_probe PRIVATE cxx_std_17)
    target_link_libraries(native_input_probe PRIVATE user32 gdi32 windowscodecs ole32 uuid)
    if(MINGW)
        target_link_options(native_input_probe PRIVATE -municode)
    endif()
endif()
