cmake_minimum_required(VERSION 3.16)
project(fc2vr_tracking_tests LANGUAGES CXX)

enable_testing()
add_executable(tracking_frame_tests test_tracking_frame.cpp)
target_compile_features(tracking_frame_tests PRIVATE cxx_std_17)
target_include_directories(tracking_frame_tests PRIVATE ../src)
if(MSVC)
    target_compile_options(tracking_frame_tests PRIVATE /W4 /WX)
else()
    target_compile_options(tracking_frame_tests PRIVATE -Wall -Wextra -Werror)
endif()

foreach(group position direction rotation stereo affine)
    add_test(NAME tracking_${group} COMMAND tracking_frame_tests ${group})
endforeach()

add_executable(pose_frame_tests test_pose_frame.cpp)
target_compile_features(pose_frame_tests PRIVATE cxx_std_17)
target_include_directories(pose_frame_tests PRIVATE ../src)
if(MSVC)
    target_compile_options(pose_frame_tests PRIVATE /W4 /WX)
else()
    target_compile_options(pose_frame_tests PRIVATE -Wall -Wextra -Werror)
endif()
foreach(group bob yaw policy once inverse bone)
    add_test(NAME pose_${group} COMMAND pose_frame_tests ${group})
endforeach()
add_test(NAME pose_source_wiring COMMAND ${CMAKE_COMMAND}
    -DSOURCE_DIR=${CMAKE_CURRENT_SOURCE_DIR}/../src -P ${CMAKE_CURRENT_SOURCE_DIR}/check_pose_wiring.cmake)

add_executable(authority_tests test_pose_authority.cpp)
target_compile_features(authority_tests PRIVATE cxx_std_17)
target_include_directories(authority_tests PRIVATE ../src)
if(MSVC)
    target_compile_options(authority_tests PRIVATE /W4 /WX)
else()
    target_compile_options(authority_tests PRIVATE -Wall -Wextra -Werror)
endif()
foreach(group freshness cache rearm cursor safety finite)
    add_test(NAME authority_${group} COMMAND authority_tests ${group})
endforeach()
add_test(NAME authority_source_wiring COMMAND ${CMAKE_COMMAND}
    -DSOURCE_DIR=${CMAKE_CURRENT_SOURCE_DIR}/../src -P ${CMAKE_CURRENT_SOURCE_DIR}/check_authority_wiring.cmake)
